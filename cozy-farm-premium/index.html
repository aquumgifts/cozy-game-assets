Use it in Godot 4

Set Project Settings → Rendering → Textures → Default Texture Filter to Nearest for crisp pixels. Animations are horizontal strips (the number before f.png is the frame count); this turns one into an AnimatedSprite2D:

var sheet: Texture2D = load("res://cozy-farm-premium/cozy-farm-portraits/portraits/1x/baker-talk-2f.png")
var frames := SpriteFrames.new()
var w := sheet.get_width() / 2
for i in 2:
	var frame := AtlasTexture.new()
	frame.atlas = sheet
	frame.region = Rect2(i * w, 0, w, sheet.get_height())
	frames.add_frame("default", frame)
$AnimatedSprite2D.sprite_frames = frames
$AnimatedSprite2D.play("default")
